1: /*********************************************************************************************************************
2:  * @file    usart.c
3:  * @brief   串口外设初始化和回调分发实现。
4:  *
5:  * @details
6:  * 1. 本文件负责：配置 USART1/2/3 并按实例分发 HAL 回调。
7:  * 2. 本文件不负责：解析蓝牙、VOFA 或舵机协议。
8:  * 3. 主要流程：HAL 回调 -> 对应模块接收/发送/错误入口。
9:  *********************************************************************************************************************/
10: 
11: #include "usart.h"
12: #include "bsp_bluetooth.h"
13: #include "feetech_sts_bus.h"
14: #include "vofa_protocol.h"
15: 
16: UART_HandleTypeDef huart1;      /* USART1 句柄：蓝牙 JDY-3X 通信。 */
17: UART_HandleTypeDef huart2;      /* USART2 句柄：ST 舵机双线 UART 通信。 */
18: UART_HandleTypeDef huart3;      /* USART3 句柄：USB-TTL 上位机通信。 */
19: 
20: //初始化 USART1 蓝牙串口。
21: void MX_USART1_UART_Init(void)
22: {
23:   huart1.Instance = USART1;                      
24:   huart1.Init.BaudRate = 9600;                      // 蓝牙波特率：9600
25:   huart1.Init.WordLength = UART_WORDLENGTH_8B;      // 8 位长度
26:   huart1.Init.StopBits = UART_STOPBITS_1;           // 停止位：1 位
27:   huart1.Init.Parity = UART_PARITY_NONE;            // 校验位：无校验
28:   huart1.Init.Mode = UART_MODE_TX_RX;               // 工作模式：发送和接收
29:   huart1.Init.HwFlowCtl = UART_HWCONTROL_NONE;      // 硬件流控：关闭
30:   huart1.Init.OverSampling = UART_OVERSAMPLING_16;  // 过采样倍数：16 倍
31: 
32:   /* HAL 初始化失败 -> 统一错误处理。 */
33:   if (HAL_UART_Init(&huart1) != HAL_OK)
34:   {
35:     Error_Handler();                                // 初始化失败时进入统一错误处理。
36:   }
37: }
38: 
39: //初始化 USART2 舵机总线串口。
40: void MX_USART2_UART_Init(void)
41: {
42:   huart2.Instance = USART2;                      
43:   huart2.Init.BaudRate = 1000000;                   // ST 舵机通信波特率：1Mbps。
44:   huart2.Init.WordLength = UART_WORDLENGTH_8B;      // 8 位长度
45:   huart2.Init.StopBits = UART_STOPBITS_1;           // 停止位：1 位。
46:   huart2.Init.Parity = UART_PARITY_NONE;            // 校验位：无校验。
47:   huart2.Init.Mode = UART_MODE_TX_RX;               // 普通 UART 双线收发。
48:   huart2.Init.HwFlowCtl = UART_HWCONTROL_NONE;      // 硬件流控：关闭。
49:   huart2.Init.OverSampling = UART_OVERSAMPLING_16;  // 过采样倍数：16 倍。
50: 
51:   if (HAL_UART_Init(&huart2) != HAL_OK)
52:   {
53:     Error_Handler();
54:   }
55: }
56: 
57: //初始化 USART3 USB-TTL 串口。
58: void MX_USART3_UART_Init(void)
59: {
60:   huart3.Instance = USART3;                        
61:   huart3.Init.BaudRate = 115200;                    // USB-TTL 上位机通信波特率：115200。
62:   huart3.Init.WordLength = UART_WORDLENGTH_8B;      // 8 位长度
63:   huart3.Init.StopBits = UART_STOPBITS_1;           // 停止位：1 位。
64:   huart3.Init.Parity = UART_PARITY_NONE;            // 校验位：无校验。
65:   huart3.Init.Mode = UART_MODE_TX_RX;               // 工作模式：发送和接收
66:   huart3.Init.HwFlowCtl = UART_HWCONTROL_NONE;      // 硬件流控：关闭。
67:   huart3.Init.OverSampling = UART_OVERSAMPLING_16;  // 过采样倍数：16 倍。
68: 
69:   if (HAL_UART_Init(&huart3) != HAL_OK)
70:   {
71:     Error_Handler();
72:   }
73: }
